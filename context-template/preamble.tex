% Note Box
\defineframedtext[NoteFrame][
    frame=on,
    framecolor=black,
    background=color,
    backgroundcolor=NoteGray,
    corner=round,
    radius=2mm,
    width=local,
    offset=1em
]

% 2. Define the automation wrapper
\unexpanded\def\startnote{%
    \startNoteFrame       % Start the visual box
    {\bf Note:-}          % Insert the title
    \par                  % Insert the break
    % The user's text will follow here automatically
}

\unexpanded\def\stopnote{%
    \stopNoteFrame        % Close the visual box
}
