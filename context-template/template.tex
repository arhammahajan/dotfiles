% --- Custom Graphics (MetaPost) ---
% This draws a shape with rounded top corners and sharp bottom corners
\startuseMPgraphic{RoundTop}
    path p ; p := unitsquare xyscaled (OverlayWidth, OverlayHeight) ;
    numeric r ; r := 3mm ;
    % Draw: Bottom-Left (sharp) -- Bottom-Right (sharp) -- Top-Right (round) -- Top-Left (round)
    fill (0,0) -- (OverlayWidth, 0) -- (OverlayWidth, OverlayHeight - r) 
         .. (OverlayWidth - r, OverlayHeight) -- (r, OverlayHeight) 
         .. (0, OverlayHeight - r) -- cycle withcolor \MPcolor{myqsfr} ;
\stopuseMPgraphic

\defineoverlay[RoundTopOverlay][\useMPgraphic{RoundTop}]

% --- Custom Visual Boxes ---

% 1. Question Box Frame (Outer container)
\defineframedtext[QuestionBoxFrame][
    frame=on,
    framecolor=myqsfr,
    rulethickness=1pt,
    corner=round,
    radius=3mm,
    width=local,
    offset=none,       % No padding so header touches the border
    background=white,
]

\starttexdefinition StartQuestionBox
    \startQuestionBoxFrame
    % The Title Bar using the custom MP background
    \framed[
        width=local,
        frame=off,
        background=RoundTopOverlay, % <--- Uses the custom shape here
        foregroundcolor=white,
        align=flushleft,
        offset=0.5em,
        strut=yes
    ]{\bf Question 1: Very important question}
    % The Content Body
    \startframed[
        width=local,
        frame=off,
        offset=1em,
        align=flushleft
    ]
\stoptexdefinition

\starttexdefinition StopQuestionBox
    \stopframed
    \stopQuestionBoxFrame
\stoptexdefinition
